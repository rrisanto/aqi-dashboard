<!DOCTYPE html>

<html>

<head>

    <title>Dashboard PM2.5</title>

    <script src="https://cdn.jsdelivr.net/npm/chart.js"></script>

</head>

<body>

    <h2>Prediksi vs Aktual PM2.5</h2>

    <canvas id="pmChart"></canvas>


    <script>

    fetch('/predict', {method: 'POST'})

        .then(response => response.json())

        .then(data => {

            const labels = data.map(d => d.Datetime);

            const actual = data.map(d => d["PM2.5"]);

            const predicted = data.map(d => d.Predicted_PM25);


            const ctx = document.getElementById('pmChart').getContext('2d');

            new Chart(ctx, {

                type: 'line',

                data: {

                    labels: labels,

                    datasets: [

                        { label: 'Actual', data: actual, borderColor: 'blue', fill: false },

                        { label: 'Predicted', data: predicted, borderColor: 'red', fill: false }

                    ]

                },

                options: { responsive: true }

            });

        });

    </script>

</body>

</html>
